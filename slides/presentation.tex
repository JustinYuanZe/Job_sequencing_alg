\section{1. Problem Statement}
\begin{frame}{Problem Definition}
    \textbf{Input:} A set of $n$ jobs $S = \{J_1, J_2, \dots, J_n\}$. Each job $J_i$ has:
    \begin{itemize}
        \item A deadline $d_i \ge 1$
        \item A profit $p_i \ge 0$
    \end{itemize}
    \vspace{0.5cm}
    \textbf{Constraints:}
    \begin{itemize}
        \item Each job takes exactly 1 unit of time to complete.
        \item Only one job can be processed at a time.
        \item A job yields profit only if completed on or before its deadline.
    \end{itemize}
    \vspace{0.5cm}
    \textbf{Objective:} Find a subset of jobs that maximizes total profit.
\end{frame}

\section{2. Mathematical Foundation}
\begin{frame}{Why Greedy Works? (The Matroid Perspective)}
    \begin{block}{The Greedy-Choice Property}
        To maximize profit, we should prioritize jobs with the highest profit. If the most profitable job can be legally scheduled, it \textbf{must} belong to the optimal solution.
    \end{block}
    \vspace{0.5cm}


\end{frame}

\section{3. Algorithmic Design \& Implementation}
\begin{frame}{Approach 1: The Naive Greedy Algorithm}
    \begin{algorithm}[H]
    \caption{Naive Job Sequencing}
    \SetAlgoLined
    Sort jobs in descending order of profit\;
    Initialize an array \texttt{slots} of size $n$ to \texttt{0}\;
    \For{each job $(p_i, d_i)$ in sorted jobs}{
        $start \leftarrow \min(n, d_i) - 1$\;
        \For{$j \leftarrow start$ \textbf{down to} 0}{
            \If{\texttt{slots[j]} = 0}{
                \texttt{slots[j]} $\leftarrow$ 1\;
                Add profit $p_i$ to Total\;
                \textbf{break}\;
            }
        }
    }
    \end{algorithm}
    \textbf{Bottleneck:} The inner loop. Worst-case Time Complexity is $\mathcal{O}(N^2)$.
\end{frame}

\begin{frame}{Approach 3: Disjoint Set (Union-Find) Optimization}
    We can eliminate the $\mathcal{O}(N)$ inner loop by treating time slots as a \textbf{Disjoint Set}.
    \begin{itemize}
        \item Each time slot is a node in a forest.
        \item The parent of node $i$ represents the \textit{greatest available time slot} $\le i$.
        \item \textbf{Find($x$):} Finds the root of $x$ (the available slot).
        \item \textbf{Union($u, v$):} Merges two sets when a slot is taken.
    \end{itemize}
    \vspace{0.5cm}
    Using \textbf{Path Compression}, the operation takes $\mathcal{O}(\alpha(N))$ time, reducing overall complexity to $\mathcal{O}(N \log N)$ (or $O(N \log D)$ depending on max deadline).
\end{frame}

\begin{frame}{Complexity Analysis}

            \bottomrule

    \vspace{0.5cm}
    \textbf{Key Observations:}
    \begin{itemize}
        \item Naive approach scales quadratically — impractical beyond $N \approx 10^4$.
        \item MinHeap and DSU both achieve near-linear performance.
        \item DSU is preferable when max deadline $D$ is small relative to $N$.
    \end{itemize}
\end{frame}
