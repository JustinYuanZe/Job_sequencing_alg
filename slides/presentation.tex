\section{1. Problem Statement}
\begin{frame}{Problem Definition}
    \textbf{Input:} A set of $n$ jobs $S = \{J_1, J_2, \dots, J_n\}$. Each job $J_i$ has:
    \begin{itemize}
        \item A deadline $d_i \ge 1$
        \item A profit $p_i \ge 0$
    \end{itemize}
    \vspace{0.5cm}
    \textbf{Constraints:}
    \begin{itemize}
        \item Each job takes exactly 1 unit of time to complete.
        \item Only one job can be processed at a time.
        \item A job yields profit only if completed on or before its deadline.
    \end{itemize}
    \vspace{0.5cm}
    \textbf{Objective:} Find a subset of jobs that maximizes total profit.
\end{frame}

\section{2. Mathematical Foundation}
\begin{frame}{Why Greedy Works? (The Matroid Perspective)}
    \begin{block}{Definition of a Matroid}
        A matroid is an ordered pair $M = (S, \mathcal{I})$ satisfying:
        \begin{enumerate}
            \item $S$ is a finite set of elements (the jobs).
            \item $\mathcal{I}$ is a non-empty family of subsets of $S$ (called independent sets) such that if $A \in \mathcal{I}$ and $B \subseteq A$, then $B \in \mathcal{I}$ (Hereditary property).
            \item If $A, B \in \mathcal{I}$ and $|A| < |B|$, there exists $x \in B \setminus A$ such that $A \cup \{x\} \in \mathcal{I}$ (Exchange property).
        \end{enumerate}
    \end{block}
    \begin{block}{Job Sequencing as a Matroid}
        Let $I \subseteq S$ be independent if the jobs in $I$ can be scheduled without violating any deadlines. It can be proved that $(S, \mathcal{I})$ is a matroid. Thus, sorting jobs by profit and greedily scheduling them guarantees the optimal solution.
    \end{block}
\end{frame}

\section{3. Algorithmic Design \& Implementation}
\begin{frame}{Approach 1: The Naive Greedy Algorithm}
    \begin{algorithm}[H]
    \caption{Naive Job Sequencing}
    \SetAlgoLined
    Sort jobs in descending order of profit\;
    Initialize an array \texttt{slots} of size $n$ to \texttt{0}\;
    \For{each job $(p_i, d_i)$ in sorted jobs}{
        $start \leftarrow \min(n, d_i) - 1$\;
        \For{$j \leftarrow start$ \textbf{down to} 0}{
            \If{\texttt{slots[j]} = 0}{
                \texttt{slots[j]} $\leftarrow$ 1\;
                Add profit $p_i$ to Total\;
                \textbf{break}\;
            }
        }
    }
    \end{algorithm}
    \textbf{Bottleneck:} The inner loop. Worst-case Time Complexity is $\mathcal{O}(N^2)$.
\end{frame}

\begin{frame}{Approach 2: Priority Queue (MinHeap) Optimization}
    \textbf{Core Idea:}
    \begin{itemize}
        \item Sort jobs in ascending order of their deadlines: $d_1 \le d_2 \le \dots \le d_n$.
        \item Maintain a Min-Heap (priority queue) of profits of the scheduled jobs.
        \item For each job, if its deadline is greater than the size of the heap, we can safely schedule it and push its profit.
        \item Otherwise, if its profit is greater than the smallest profit currently scheduled (the top of the heap), we pop the top job and schedule the new one instead.
    \end{itemize}
    \vspace{0.5cm}
    \textbf{Complexity Improvement:} Worst-case time is reduced to $\mathcal{O}(N \log N)$ and space is $\mathcal{O}(N)$.
\end{frame}

\begin{frame}{Approach 3: Disjoint Set (Union-Find) Optimization}
    We can eliminate the $\mathcal{O}(N)$ inner loop by treating time slots as a \textbf{Disjoint Set}.
    \begin{itemize}
        \item Each time slot $0 \dots D$ is represented as a node in a forest.
        \item The representative element of node $i$ (found via \texttt{Find($i$)}) points to the \textit{latest available time slot} $\le i$.
        \item \textbf{Find($x$):} Finds the free slot for deadline $x$ using path compression.
        \item \textbf{Union($u, v$):} Once slot $v$ is occupied, we merge its set with slot $v-1$, so future jobs with deadline $v$ look at $v-1$ or earlier.
    \end{itemize}
    \vspace{0.3cm}
    With path compression, each find operation runs in nearly $\mathcal{O}(\alpha(D))$. The overall complexity is $\mathcal{O}(N \log N + N \alpha(D))$, dominated by the initial sort.
\end{frame}

\begin{frame}{Approach 4: Optimized Radix DSU (Proposed Optimization)}
    \textbf{The Sorting Bottleneck:}
    Both DSU and Min-Heap require sorting, costing $\mathcal{O}(N \log N)$.
    \vspace{0.3cm}
    
    \textbf{Our Linear-Time Solution:}
    \begin{itemize}
        \item Since profits are positive integers, we can use a stable \textbf{Radix Sort} (using base 256) to sort all jobs by profit in $\mathcal{O}(N)$ time (4 passes for 32-bit integers).
        \item Once sorted, we use DSU with path compression to find available slots in $\mathcal{O}(N \alpha(D))$ time.
        \item \textbf{Overall Complexity:} $\mathcal{O}(N + N \alpha(D))$ -- practically \textbf{linear time}!
    \end{itemize}
\end{frame}

\section{4. Performance Analysis}

\begin{frame}{Complexity Analysis}
    \begin{table}[]
        \centering
        \begin{tabular}{lcc}
            \toprule
            \textbf{Algorithm} & \textbf{Time Complexity} & \textbf{Space Complexity} \\
            \midrule
            Naive Array & $\mathcal{O}(N^2)$ & $\mathcal{O}(N)$ \\
            MinHeap (Expected) & $\mathcal{O}(N \log N)$ & $\mathcal{O}(N)$ \\
            Disjoint Set (Optimal) & $\mathcal{O}(N \log D)$ & $\mathcal{O}(D)$ \\
            \bottomrule
        \end{tabular}
    \end{table}

    \vspace{0.5cm}
    \textbf{Key Observations:}
    \begin{itemize}
        \item Naive approach scales quadratically — impractical beyond $N \approx 10^4$.
        \item MinHeap and DSU both achieve near-linear performance.
        \item DSU is preferable when max deadline $D$ is small relative to $N$.
    \end{itemize}
\end{frame}

\section{5. Real-World Applications \& Extensions}

\begin{frame}{Extension 1: Two-Machine Scheduling (Johnson's Algorithm)}
    \textbf{Beyond Single Machine:}
    \begin{itemize}
        \item Shifting from maximizing profit under deadlines on 1 machine to minimizing total completion time (\textbf{makespan}) across 2 machines in series ($M_1 \to M_2$).
        \item \textbf{Johnson's Rule ($O(N \log N)$):}
            \begin{itemize}
                \item Group 1 ($t_{1i} < t_{2i}$): sort ascending by $t_{1i}$, scheduled first.
                \item Group 2 ($t_{1i} \ge t_{2i}$): sort descending by $t_{2i}$, scheduled last.
            \end{itemize}
        \item \textbf{Real-world Complexity:} If machines have planned maintenance windows (Machine Unavailability), makespan minimization becomes NP-hard (Rapine et al., 2013).
    \end{itemize}
\end{frame}

\begin{frame}{Extension 2: Explainable AI (XAI) in Industrial Sequencing}
    \textbf{Bridging Algorithms and Human Operations:}
    \begin{itemize}
        \item Factory floor managers often reject automated schedules because they are "black-boxes" (e.g., "Why is Job 5 delayed?").
        \item \textbf{Explainable AI (XAI) Solutions:}
            \begin{itemize}
                \item \textbf{Feature Attribution (SHAP/LIME):} Quantifies the impact of a job's profit, deadline, or machine requirements on its priority.
                \item \textbf{Counterfactual Explanations:} Tells operators exactly what to modify: \textit{"If Job B's profit increases by 15\% or its deadline is extended by 2 hours, it will be processed today."}
            \end{itemize}
    \end{itemize}
\end{frame}

\section{6. Conclusion}
\begin{frame}{Conclusion}
    \begin{itemize}
        \item Job Sequencing is a prime example of how understanding the underlying mathematics (Matroids) justifies the Greedy approach.
        \item While basic logic solves the problem ($\mathcal{O}(N^2)$), advanced data structures (MinHeap, Disjoint Set) are required to scale the solution ($\mathcal{O}(N \log N)$).
        \item \textbf{Future Scope:} Real-world constraints (like multi-machine setups and need for explainability) require expanding these classic algorithms into heuristic and AI-driven domains.
    \end{itemize}
