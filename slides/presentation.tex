\section{1. Problem Statement}
\begin{frame}{Problem Definition}
    \textbf{Input:} A set of $n$ jobs $S = \{J_1, J_2, \dots, J_n\}$. Each job $J_i$ has:
    \begin{itemize}
        \item A deadline $d_i \ge 1$
        \item A profit $p_i \ge 0$
    \end{itemize}
    \vspace{0.5cm}
    \textbf{Constraints:}
    \begin{itemize}
        \item Each job takes exactly 1 unit of time to complete.
        \item Only one job can be processed at a time.
        \item A job yields profit only if completed on or before its deadline.
    \end{itemize}
    \vspace{0.5cm}
    \textbf{Objective:} Find a subset of jobs that maximizes total profit.
\end{frame}

\section{2. Mathematical Foundation}
\begin{frame}{Why Greedy Works? (The Matroid Perspective)}
    \begin{block}{Definition of a Matroid}
        A matroid is an ordered pair $M = (S, \mathcal{I})$ satisfying:
        \begin{enumerate}
            \item $S$ is a finite set of elements (the jobs).
            \item $\mathcal{I}$ is a non-empty family of subsets of $S$ (called independent sets) such that if $A \in \mathcal{I}$ and $B \subseteq A$, then $B \in \mathcal{I}$ (Hereditary property).
            \item If $A, B \in \mathcal{I}$ and $|A| < |B|$, there exists $x \in B \setminus A$ such that $A \cup \{x\} \in \mathcal{I}$ (Exchange property).
        \end{enumerate}
    \end{block}
    \begin{block}{Job Sequencing as a Matroid}
        Let $I \subseteq S$ be independent if the jobs in $I$ can be scheduled without violating any deadlines. It can be proved that $(S, \mathcal{I})$ is a matroid. Thus, sorting jobs by profit and greedily scheduling them guarantees the optimal solution.
    \end{block}
\end{frame}

\section{3. Algorithmic Design \& Implementation}

\begin{frame}{Approach 1: The Naive Greedy Algorithm}
    \begin{minipage}[t]{0.64\textwidth}
        \begin{algorithm}[H]
        \scriptsize
        \caption{Naive Greedy}
        \SetAlgoLined
        $n \leftarrow \text{length}(deadline)$\;
        $countJobs \leftarrow 0$, $totalProfit \leftarrow 0$, $jobs \leftarrow \text{empty list}$\;
        \For{$i \leftarrow 0$ \text{to} $n - 1$}{
            $jobs.\text{add}((profit[i], deadline[i]))$\;
        }
        sort $jobs$ in descending order by profit\;
        $slot \leftarrow \text{array of size } n \text{ initialized with } 0$\;
        \For{each $job$ in $jobs$}{
            $latestSlot \leftarrow \min(n, job.deadline) - 1$\;
            \For{$j \leftarrow latestSlot$ \textbf{down to} 0}{
                \If{$slot[j] = 0$}{
                    $slot[j] \leftarrow 1$\;
                    $countJobs \leftarrow countJobs + 1$\;
                    $totalProfit \leftarrow totalProfit + job.profit$\;
                    \textbf{break}\;
                }
            }
        }
        \Return $countJobs, totalProfit$\;
        \end{algorithm}
    \end{minipage}
    \hfill
    \begin{minipage}[t]{0.33\textwidth}
        \vspace{0.4cm}
        \small
        \textbf{Time Complexity:} \\ $\mathcal{O}(N^2)$ due to backward scan in nested loop. \\
        \vspace{0.5cm}
        \textbf{Space Complexity:} \\ $\mathcal{O}(N)$ to store slots.
    \end{minipage}
\end{frame}

\begin{frame}{Approach 2: Priority Queue (MinHeap) Optimization}
    \textbf{Core Idea:}
    \begin{itemize}
        \item Sort jobs in ascending order of their deadlines: $d_1 \le d_2 \le \dots \le d_n$.
        \item Maintain a Min-Heap of the profits of scheduled jobs.
        \item For each job, if its deadline is greater than the size of the heap, we can schedule it and push its profit.
        \item Otherwise, if its profit is greater than the minimum profit currently scheduled (heap top), replace the top job.
    \end{itemize}
    \vspace{0.3cm}
    \textbf{Complexity Improvement:} Worst-case time is reduced to $\mathcal{O}(N \log N)$ and space is $\mathcal{O}(N)$.
\end{frame}

\begin{frame}{Approach 2: Min-Heap Optimization Pseudocode}
    \begin{algorithm}[H]
    \tiny
    \caption{Min-Heap Greedy}
    \SetAlgoLined
    $n \leftarrow \text{length}(deadline)$\;
    $jobCount \leftarrow 0$, $totalProfit \leftarrow 0$, $jobs \leftarrow \text{empty list}$\;
    \For{$i \leftarrow 0$ \text{to} $n - 1$}{
        $jobs.\text{add}((deadline[i], profit[i]))$\;
    }
    sort $jobs$ in ascending order by deadline\;
    $pq \leftarrow \text{empty min-heap}$\;
    \For{each $job$ in $jobs$}{
        \If{$job.deadline > \text{size}(pq)$}{
            $pq.\text{insert}(job.profit)$\;
        }
        \ElseIf{$pq$ is not empty \textbf{and} $pq.\text{minimum}() < job.profit$}{
            $pq.\text{removeMinimum}()$\;
            $pq.\text{insert}(job.profit)$\;
        }
    }
    \While{$pq$ is not empty}{
        $totalProfit \leftarrow totalProfit + pq.\text{minimum}()$\;
        $pq.\text{removeMinimum}()$\;
        $jobCount \leftarrow jobCount + 1$\;
    }
    \Return $jobCount, totalProfit$\;
    \end{algorithm}
\end{frame}

\begin{frame}{Approach 3: Disjoint Set (Union-Find) Helper Operations}
    \begin{algorithm}[H]
    \footnotesize
    \caption{Disjoint Set Helper Operations}
    \SetAlgoLined
    \SetKwFunction{FMake}{MAKE-SET}
    \SetKwFunction{FFind}{FIND}
    \SetKwFunction{FMerge}{MERGE}
    \SetKwProg{Fn}{Function}{:}{}

    \Fn{\FMake{n}}{
        create parent array of size $n+1$\;
        \For{$i \leftarrow 0$ \text{to} $n$}{
            $parent[i] \leftarrow i$\;
        }
        \Return $parent$\;
    }
    \Fn{\FFind{s}}{
        \If{$parent[s] = s$}{
            \Return $s$\;
        }
        $parent[s] \leftarrow \FFind{parent[s]}$ \quad \text{// Path Compression}\;
        \Return $parent[s]$\;
    }
    \Fn{\FMerge{u, v}}{
        $parent[v] \leftarrow u$\;
    }
    \end{algorithm}
\end{frame}

\begin{frame}{Approach 3: Disjoint Set (Union-Find) Pseudocode}
    \begin{algorithm}[H]
    \scriptsize
    \caption{DSU Greedy}
    \SetAlgoLined
    $n \leftarrow \text{length}(deadline)$\;
    $jobCount \leftarrow 0$, $totalProfit \leftarrow 0$, $jobs \leftarrow \text{empty list}$\;
    \For{$i \leftarrow 0$ \text{to} $n - 1$}{
        $jobs.\text{add}((profit[i], deadline[i]))$\;
    }
    sort $jobs$ in descending order by profit\;
    $maxDeadline \leftarrow \text{maximum value in } deadline$\;
    create DisjointSet $ds$ with $maxDeadline$ slots\;
    \For{each $job$ in $jobs$}{
        $availableSlot \leftarrow ds.\text{find}(job.deadline)$\;
        \If{$availableSlot > 0$}{
            $ds.\text{merge}(ds.\text{find}(availableSlot - 1), availableSlot)$\;
            $totalProfit \leftarrow totalProfit + job.profit$\;
            $jobCount \leftarrow jobCount + 1$\;
        }
    }
    \Return $jobCount, totalProfit$\;
    \end{algorithm}
\end{frame}

\begin{frame}{Approach 4: Optimized Radix DSU (Proposed Optimization)}
    \textbf{The Sorting Bottleneck:}
    Both DSU and Min-Heap require sorting, costing $\mathcal{O}(N \log N)$.
    \vspace{0.3cm}

    \textbf{Our Linear-Time Solution:}
    \begin{itemize}
        \item Since profits are positive integers, we can use a stable \textbf{Radix Sort} (using base 256) to sort all jobs by profit in $\mathcal{O}(N)$ time (4 passes for 32-bit integers).
        \item Once sorted, we use DSU with path compression to find available slots in $\mathcal{O}(N \alpha(D))$ time.
        \item \textbf{Overall Complexity:} $\mathcal{O}(N + N \alpha(D))$ -- practically \textbf{linear time}!
    \end{itemize}
\end{frame}

\begin{frame}{Approach 4: Radix Sort Helper Pseudocode}
    \vspace{-0.25cm}
    \begin{algorithm}[H]
    \tiny
    \caption{Radix Sort Jobs Descending}
    \SetAlgoLined
    $n \leftarrow \text{size}(jobs)$\;
    \If{$n \le 1$}{
        \Return $jobs$\;
    }
    $maxProfit \leftarrow 0$\;
    \For{each $job$ in $jobs$}{
        $maxProfit \leftarrow \max(maxProfit, job.profit)$\;
    }
    $temp \leftarrow \text{array of size } n$\;
    \For{$shift \leftarrow 0$; $(maxProfit \gg shift) > 0$; $shift \leftarrow shift + 8$}{
        $count[0..255] \leftarrow 0$\;
        \For{each $job$ in $jobs$}{
            $byteVal \leftarrow (job.profit \gg shift) \& 255$\;
            $count[byteVal] \leftarrow count[byteVal] + 1$\;
        }
        \For{$i \leftarrow 1$ \textbf{to} $255$}{
            $count[i] \leftarrow count[i] + count[i - 1]$\;
        }
        \For{$i \leftarrow n - 1$ \textbf{down to} $0$}{
            $byteVal \leftarrow (jobs[i].profit \gg shift) \& 255$\;
            $temp[count[byteVal] - 1] \leftarrow jobs[i]$\;
            $count[byteVal] \leftarrow count[byteVal] - 1$\;
        }
        $jobs \leftarrow temp$\;
    }
    reverse $jobs$ to obtain descending profit order\;
    \Return $jobs$\;
    \end{algorithm}
\end{frame}

\begin{frame}{Approach 4: Radix-DSU Pseudocode}
    \begin{algorithm}[H]
    \scriptsize
    \caption{Radix-DSU Greedy}
    \SetAlgoLined
    $n \leftarrow \text{length}(deadline)$\;
    $jobCount \leftarrow 0$, $totalProfit \leftarrow 0$, $jobs \leftarrow \text{empty list}$\;
    \For{$i \leftarrow 0$ \textbf{to} $n - 1$}{
        $jobs.\text{add}((profit[i], deadline[i]))$\;
    }
    $jobs \leftarrow \text{RADIX-SORT-JOBS-DESCENDING}(jobs)$\;
    $maxDeadline \leftarrow \text{maximum value in } deadline$\;
    create DisjointSet $ds$ with $maxDeadline$ slots\;
    \For{each $job$ in $jobs$}{
        $availableSlot \leftarrow ds.\text{find}(job.deadline)$\;
        \If{$availableSlot > 0$}{
            $ds.\text{merge}(ds.\text{find}(availableSlot - 1), availableSlot)$\;
            $totalProfit \leftarrow totalProfit + job.profit$\;
            $jobCount \leftarrow jobCount + 1$\;
        }
    }
    \Return $jobCount, totalProfit$\;
    \end{algorithm}
\end{frame}

\section{4. Performance Analysis}

\begin{frame}{Complexity Analysis}
    \begin{table}[]
        \centering
        \begin{tabular}{lcc}

            \midrule

        \end{tabular}

    \vspace{0.5cm}
    \textbf{Key Observations:}
    \begin{itemize}
        \item Naive approach scales quadratically — impractical beyond $N \approx 10^4$.
        \item MinHeap and DSU both achieve near-linear performance.
        \item DSU is preferable when max deadline $D$ is small relative to $N$.
    \end{itemize}

\section{5. Real-World Applications \& Extensions}

\begin{frame}{Extension 1: Two-Machine Scheduling (Johnson's Algorithm)}
    \textbf{Beyond Single Machine:}
    \begin{itemize}
        \item Shifting from maximizing profit under deadlines on 1 machine to minimizing total completion time (\textbf{makespan}) across 2 machines in series ($M_1 \to M_2$).
        \item \textbf{Johnson's Rule ($O(N \log N)$):}
            \begin{itemize}
                \item Group 1 ($t_{1i} < t_{2i}$): sort ascending by $t_{1i}$, scheduled first.
                \item Group 2 ($t_{1i} \ge t_{2i}$): sort descending by $t_{2i}$, scheduled last.
            \end{itemize}
        \item \textbf{Real-world Complexity:} If machines have planned maintenance windows (Machine Unavailability), makespan minimization becomes NP-hard (Rapine et al., 2013).
    \end{itemize}
\end{frame}

\begin{frame}{Extension 2: Explainable AI (XAI) in Industrial Sequencing}
    \textbf{Bridging Algorithms and Human Operations:}
    \begin{itemize}
        \item Factory floor managers often reject automated schedules because they are "black-boxes" (e.g., "Why is Job 5 delayed?").
        \item \textbf{Explainable AI (XAI) Solutions:}
            \begin{itemize}
                \item \textbf{Feature Attribution (SHAP/LIME):} Quantifies the impact of a job's profit, deadline, or machine requirements on its priority.
                \item \textbf{Counterfactual Explanations:} Tells operators exactly what to modify: \textit{"If Job B's profit increases by 15\% or its deadline is extended by 2 hours, it will be processed today."}
            \end{itemize}
    \end{itemize}
\end{frame}

\section{6. Conclusion}
\begin{frame}{Conclusion}
    \begin{itemize}
        \item Job Sequencing is a prime example of how understanding the underlying mathematics (Matroids) justifies the Greedy approach.
        \item While basic logic solves the problem ($\mathcal{O}(N^2)$), advanced data structures (MinHeap, Disjoint Set) are required to scale the solution ($\mathcal{O}(N \log N)$).
        \item \textbf{Future Scope:} Real-world constraints (like multi-machine setups and need for explainability) require expanding these classic algorithms into heuristic and AI-driven domains.
    \end{itemize}
